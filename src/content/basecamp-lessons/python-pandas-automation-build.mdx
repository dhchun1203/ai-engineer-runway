```python
# 준비 칸: 노트를 열 때마다 맨 처음 한 번 실행하세요
import csv
from pathlib import Path

Path("data").mkdir(exist_ok=True)          # data 폴더 만들기
Path("추가파일").mkdir(exist_ok=True)       # 추가파일 폴더 만들기

    sheet.append(header)
    for row in rows:

# 종로점: 깔끔한 엑셀 파일
```

## 1단계. STEP 4 코드 실행하기: 이게 목표예요

STEP 4 보강 레슨 두 편에 흩어져 있던 코드를 한 칸에 모았어요. AI에게 받아 붙였던 분야별 합계 시트만 빼고 그대로예요. 새 칸에 붙여 넣어 실행하세요.

```python
# STEP 4 버전: 서점 엑셀 자동 정리 툴
import csv
from pathlib import Path
from openpyxl import Workbook, load_workbook

DATA_FOLDER = Path("data")
RESULT_FILE = Path("주간판매_정리.xlsx")

# [A] 지저분한 값을 정리하는 함수
def to_number(value):
    return int(str(value).replace(",", "").strip())

def clean_text(value):
    return str(value).strip()

# [B] 엑셀이든 CSV든 읽는 함수
def read_file(path):
    if path.suffix == ".csv":
        with open(path, encoding="utf-8-sig") as f:
            return list(csv.DictReader(f))
    sheet = load_workbook(path).active
    rows = list(sheet.iter_rows(values_only=True))
    header = rows[0]
    return [dict(zip(header, row)) for row in rows[1:]]

# [C] 폴더 전체를 읽어서 한 줄씩 정리해 모으기
all_rows = []
errors = []
for path in sorted(DATA_FOLDER.iterdir()):
    try:
        for row in read_file(path):
            count = to_number(row["권수"])
            price = to_number(row["가격"])
            all_rows.append({
                "날짜": row["날짜"],
                "지점": clean_text(row["지점"]),
                "분야": clean_text(row["분야"]),
                "권수": count,
                "가격": price,
                "매출": count * price,
            })
        print("[읽기 성공]", path.name)
    except Exception as e:
        errors.append(path.name)
        print("[건너뜀]", path.name, "/", type(e).__name__)

# [D] 지점별 매출 합계
branch_total = {}
for row in all_rows:
    branch = row["지점"]
    branch_total[branch] = branch_total.get(branch, 0) + row["매출"]

# [E] 시트 3개짜리 결과 파일 저장
wb = Workbook()
sheet1 = wb.active
sheet1.title = "전체데이터"
sheet1.append(["날짜", "지점", "분야", "권수", "가격", "매출"])
for row in all_rows:
    sheet1.append(list(row.values()))

sheet2 = wb.create_sheet("지점별합계")
sheet2.append(["지점", "총매출"])
for branch, total in branch_total.items():
    sheet2.append([branch, total])

sheet3 = wb.create_sheet("확인필요")
sheet3.append(["열지 못한 파일"])
for name in errors:
    sheet3.append([name])

wb.save(RESULT_FILE)

# [F] 결과 확인 출력
print("모은 줄 수:", len(all_rows))
print("지점별 합계:", branch_total)
print(f"완료! 총 {len(all_rows)}줄 → {RESULT_FILE}")
```

```text
[건너뜀] 광주점.xlsx / BadZipFile
[읽기 성공] 신촌점.xlsx
[읽기 성공] 종로점.xlsx
[읽기 성공] 해운대점.csv
모은 줄 수: 9
지점별 합계: {'신촌점': 666000, '종로점': 444000, '해운대점': 451000}
완료! 총 9줄 → 주간판매_정리.xlsx
```

이 결과가 우리의 목표예요. Pandas 버전도 9줄, 같은 지점별 합계, 같은 내용의 결과 파일이 나와야 해요.

> 지난주 툴과 하나 다른 점이 있어요. 2편의 "AI에게 기능 더하기"에서 붙였던 분야별합계 시트는 뺐어요. 시트 세 개짜리 기본 툴을 번역하는 게 이번 목표라서요. 공식 과제의 STEP 4 정답 코드도 시트 세 개짜리예요.

## 4단계. [A] 정리 함수를 Pandas로: clean_df()

STEP 4의 `to_number()`와 `clean_text()`는 값 하나를 받아서 정리했어요. Pandas 버전은 지점 표(DataFrame) 하나를 통째로 받아서 정리해요.

```python
def clean_df(df):
    """ 서점 지점 표 하나를 정리한다: 공백 지우기, 글자 숫자를 숫자로, 매출 열 추가 """
    df["지점"] = df["지점"].str.strip()
    df["분야"] = df["분야"].str.strip()
    df["권수"] = df["권수"].astype(str).str.replace(",", "").astype(int)
    df["가격"] = df["가격"].astype(str).str.replace(",", "").astype(int)
    df["매출"] = df["권수"] * df["가격"]
    return df
```

[STEP 4 코드를 Pandas로 바꿔 보기](/basecamp/python-pandas-excel-automation)의 카페 `clean_df()`와 모양이 같아요. 다른 건 열 이름뿐이에요. 카페는 품목, 수량, 단가였고 서점은 분야, 권수, 가격이에요.

권수와 가격 줄 맨 앞의 `astype(str)`, 기억나죠? 종로점처럼 이미 숫자인 열에는 `.str`을 쓸 수 없어요. 그래서 일단 글자로 바꾸고 정리해요. STEP 4의 `to_number()`가 `str(value)`로 시작했던 것과 같은 이유예요.

### 함수 바로 시험해 보기

함수는 만들자마자 작은 표로 시험해 봐요. 글자 숫자와 진짜 숫자를 일부러 섞어 넣었어요.

```python
test = pd.DataFrame({

    "분야": [" 소설 ", "에세이"],
    "권수": ["20", 8],             # 글자 숫자와 진짜 숫자가 섞여 있어요
    "가격": ["15,000", 13000],

print(clean_df(test))
print(clean_df(test)["분야"].tolist())
```

## 5단계. [B] 읽기 함수를 Pandas로: read_file()

STEP 4의 `read_file()`은 CSV 부분과 엑셀 부분을 합쳐 8줄이었어요. Pandas에서는 이렇게 줄어요.

```python
def read_file(path):
    """ 엑셀이든 CSV든 DataFrame으로 읽어 온다 """
    if path.suffix == ".csv":
        return pd.read_csv(path)
    return pd.read_excel(path)
```

이름이 1단계의 `read_file()`과 같죠? 이 칸을 실행하는 순간 `read_file`은 새 Pandas 버전으로 바뀌어요. 같은 이름으로 다시 만들면 나중에 만든 쪽이 이기거든요. 1단계 칸을 다시 실행하면 거꾸로 STEP 4 버전으로 돌아가니, 칸 실행 순서를 지켜 주세요.

```text
['15,000', '13,000', '32,000']
```

해운대점은 권수가 바로 더해졌어요(15 + 10 + 3 = 28). `read_csv`가 숫자를 알아서 숫자로 읽었기 때문이에요. STEP 4의 `csv.DictReader`였다면 전부 글자라서 더하기가 안 됐을 거예요.

신촌점 가격은 아직 따옴표가 붙은 글자예요. 쉼표 때문에 숫자로 못 알아본 거죠. 이건 다음 단계에서 `clean_df()`가 정리해요. 읽기와 정리를 나눠 둔 덕분에, 각 함수가 자기 일 하나만 하면 돼요.

## 6단계. [C] 폴더 전체 읽어서 합치기

STEP 4에서는 파일 안의 줄을 하나씩 정리해서 `all_rows`에 쌓았어요. Pandas에서는 표를 통째로 정리해서 `tables`에 모은 뒤, 마지막에 한 번에 합쳐요.

```python
tables = []      # 정리된 지점 표들
errors = []      # 열지 못한 파일 목록

for path in sorted(DATA_FOLDER.iterdir()):
    try:
        tables.append(clean_df(read_file(path)))   # 읽고, 정리하고, 모으기
        print("[읽기 성공]", path.name)
    except Exception as e:
        errors.append(path.name)
        print("[건너뜀]", path.name, "/", type(e).__name__)

all_df = pd.concat(tables, ignore_index=True)       # 모은 표를 하나로
```

```text
[건너뜀] 광주점.xlsx / ValueError
[읽기 성공] 신촌점.xlsx
[읽기 성공] 종로점.xlsx
[읽기 성공] 해운대점.csv
```

STEP 4 [C]와 나란히 놓고 보세요.

| | STEP 4 [C] | Pandas [C] |
| --- | --- | --- |
| 반복문 | 두 겹 (파일, 그 안의 줄) | 한 겹 (파일만) |
| 한 번에 모으는 단위 | 한 줄 (딕셔너리) | 한 표 (DataFrame) |
| 정리하는 곳 | 반복문 안에서 값마다 `to_number()` | `clean_df()`가 열 전체를 한 번에 |
| 합치기 | 쌓는 순간 이미 한 리스트 | 마지막에 `pd.concat()` |
| 깨진 파일 | `try` / `except`로 건너뛰기 | 똑같아요 |
| 코드 줄 수 | 19줄 | 10줄 |

광주점 에러 이름이 BadZipFile에서 ValueError로 바뀐 것도 보이죠? Pandas는 파일을 열기 전에 "이건 엑셀 파일 모양이 아니다"를 먼저 알아채서 다른 에러를 내요. `except Exception`이 모든 에러를 잡으니 똑같이 건너뛰었어요.

STEP 4와 같은 9줄이에요.

- 신촌점 3줄(0, 1, 2번)이 먼저 오고, 그다음 3번이 종로점 첫 줄이에요. 광주점은 이름순으로 맨 앞이지만 건너뛰었으니까요.
- `.to_dict()`는 그 줄을 STEP 4에서 익숙한 딕셔너리 모양으로 보여 줘요. STEP 4의 `all_rows[3]`과 비교해 보세요. 똑같이 생겼을 거예요.
